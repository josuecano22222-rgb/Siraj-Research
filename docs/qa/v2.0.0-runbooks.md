## 1. Migration safety (install main → load data → upgrade → migrate)

Proves: no data loss, no duplicate registry rows, the new
`v2_0.backfill_family_derived` patch applies exactly once and only
touches derived columns.

```bash
cd ~/frappe-bench

# --- A. Fresh site on CURRENT main -------------------------------------
git -C apps/siraj_research fetch origin main
git -C apps/siraj_research reset --hard origin/main
bench new-site v2_migration_test --db-root-password root --admin-password admin
bench --site v2_migration_test install-app siraj_research

# Seed + demo data (after_install already seeds reference data; this also
# proves seeding is exists-guarded / idempotent — run it twice)
bench --site v2_migration_test execute siraj_research.setup.master_data.seed_master_data
bench --site v2_migration_test execute siraj_research.setup.master_data.seed_master_data
bench --site v2_migration_test siraj-research load-demo-data

# --- B. Snapshot BEFORE the upgrade ------------------------------------
bench --site v2_migration_test mariadb -e "
SELECT 'Person' t, COUNT(*) c FROM \`tabPerson\`
UNION ALL SELECT 'Source', COUNT(*) FROM \`tabSource\`
UNION ALL SELECT 'Event', COUNT(*) FROM \`tabEvent\`
UNION ALL SELECT 'Governorate', COUNT(*) FROM \`tabGovernorate\`
UNION ALL SELECT 'Village', COUNT(*) FROM \`tabVillage\`
UNION ALL SELECT 'Community', COUNT(*) FROM \`tabCommunity\`;"
# Duplicate-registry check (must print an empty result set):
bench --site v2_migration_test mariadb -e "
SELECT 'Governorate' t, code, COUNT(*) c FROM \`tabGovernorate\` GROUP BY code HAVING c>1
UNION ALL
SELECT 'Village', code, COUNT(*) FROM \`tabVillage\` GROUP BY code HAVING c>1;"

# --- C. Upgrade to the V2.0.0 code and migrate --------------------------
git -C apps/siraj_research fetch origin arena/e7d91cd0-siraj-research
git -C apps/siraj_research reset --hard origin/arena/e7d91cd0-siraj-research
# (post-merge: git reset --hard origin/main && git checkout V2.0.0)
bench --site v2_migration_test migrate

# --- D. Assertions AFTER the upgrade ------------------------------------
# 1) counts identical to the BEFORE snapshot:
bench --site v2_migration_test mariadb -e "
SELECT 'Person' t, COUNT(*) c FROM \`tabPerson\`
UNION ALL SELECT 'Source', COUNT(*) FROM \`tabSource\`
UNION ALL SELECT 'Event', COUNT(*) FROM \`tabEvent\`
UNION ALL SELECT 'Governorate', COUNT(*) FROM \`tabGovernorate\`
UNION ALL SELECT 'Village', COUNT(*) FROM \`tabVillage\`
UNION ALL SELECT 'Community', COUNT(*) FROM \`tabCommunity\`;"
# 2) no duplicate registry rows (empty result set):
bench --site v2_migration_test mariadb -e "
SELECT code, COUNT(*) c FROM \`tabGovernorate\` GROUP BY code HAVING c>1;"
# 3) the new patch ran exactly once:
bench --site v2_migration_test mariadb -e "
SELECT patch, COUNT(*) c FROM \`tabPatch Log\`
WHERE patch LIKE '%siraj_research%' GROUP BY patch HAVING c>1;
SELECT patch FROM \`tabPatch Log\` WHERE patch = 'siraj_research.patches.v2_0.backfill_family_derived';"
#    → one row for v2_0.backfill_family_derived; no patch logged twice.
# 4) derived family columns are populated (they were empty/0 before the
#    first save on main):
bench --site v2_migration_test mariadb -e "
SELECT name, is_grandparent, family_summary FROM \`tabPerson\` LIMIT 5;"

# --- E. Idempotency ------------------------------------------------------
bench --site v2_migration_test migrate   # second migrate: patch is in Patch Log → skipped
bench --site v2_migration_test mariadb -e "SELECT COUNT(*) FROM \`tabPerson\`;"
#    → same count; Patch Log still has exactly one v2_0 row.

# Cleanup when done:
bench --site v2_migration_test --force drop-site   # DESTRUCTIVE: deletes v2_migration_test only
```

**Pass criteria:** every AFTER count equals BEFORE; both duplicate
queries return empty; exactly one `v2_0.backfill_family_derived` row in
Patch Log; a second migrate changes nothing.

### Run locally on your bench (paste into `bench --site frappe.com console`)

Each probe prints PASS/FAIL with the expected value in the assertion.
Create the probe users once (System Manager UI or the script below):

```python
# --- one-time probe users (console) ------------------------------------
import frappe
for name, role in (("probe_viewer", "SIRAJ Viewer"),
                   ("probe_researcher", "SIRAJ Researcher")):
    if not frappe.db.exists("User", name):
        u = frappe.get_doc({"doctype": "User", "email": name + "@probe.invalid",
                            "first_name": name, "role": role})
        u.insert(ignore_permissions=True)
    else:
        frappe.db.set_value("User", name, "role", role)
frappe.db.commit()
```

**A. Guest access (must be blocked)**

```python
import frappe
frappe.set_user("Guest")
assert not frappe.has_permission("Person", "read"),  "guest CAN read Person"
assert not frappe.has_permission("Person", "write"), "guest CAN write Person"
assert not frappe.has_permission("Source", "read"),  "guest CAN read Source"
frappe.set_user("Administrator")
print("GUEST probe PASS")
```

**B. Custom role combinations (least privilege holds)**

```python
import frappe
# Viewer: read Person, cannot write.
frappe.set_user("probe_viewer")
assert frappe.has_permission("Person", "read"), "viewer cannot read"
assert not frappe.has_permission("Person", "write"), "viewer CAN write"
frappe.set_user("probe_researcher")
assert frappe.has_permission("Person", "read"), "researcher cannot read"
assert frappe.has_permission("Person", "write"), "researcher cannot write"
frappe.set_user("Administrator")
print("ROLE-COMBINATION probe PASS")
```

**C. Link-/level-restricted fields (permlevel > 0)**

```python
import frappe
# researcher_notes on Person is permlevel 1 (only Reviewer/Admin rows grant it).
meta = frappe.get_meta("Person")
field = meta.get_field("researcher_notes")
assert field and field.permlevel == 1, "researcher_notes must stay permlevel 1"
name = frappe.get_value("Person", {}, "name")
if not name:
    print("PERMLEVEL probe SKIP — no Person records yet (load demo data first)")
else:
    frappe.set_user("probe_viewer")
    doc = frappe.get_doc("Person", name)
    assert not doc.has_permlevel_access_to("researcher_notes"), "viewer HAS permlevel-1 access"
    frappe.set_user("probe_researcher")
    doc = frappe.get_doc("Person", name)
    assert not doc.has_permlevel_access_to("researcher_notes"), "researcher HAS permlevel-1 access"
    frappe.set_user("Administrator")
    doc = frappe.get_doc("Person", name)
    assert doc.has_permlevel_access_to("researcher_notes"), "admin lacks permlevel-1 access"
    print("PERMLEVEL probe PASS")
frappe.set_user("Administrator")
```

**D. Submit / cancel (workflow transitions)**

```python
import frappe
# SIRAJ DocTypes are workflow-driven (Draft → In Review → Verified → …),
# not docstatus-submittable: assert both facts, then check transition roles.
subs = [dt for dt in ("Person", "Source", "Event", "Armament")
        if frappe.get_meta(dt).is_submittable]
assert not subs, f"unexpected submittable DocType(s): {subs}"
if frappe.db.exists("Workflow", "SIRAJ Review"):
    transitions = frappe.workflow.get_transitions(
        {"doctype": "Armament", "docstatus": 0, "workflow_state": "Draft"})
    assert transitions, "SIRAJ Review exists but Draft has no transitions"
    assert all(t.get("role") for t in transitions), \
        "role-less transition found — everyone could act on Draft"
    print("SUBMIT/CANCEL probe PASS — Draft transition roles:",
          sorted({t["role"] for t in transitions}))
else:
    print("SUBMIT/CANCEL probe PASS (non-submittable); "
          "SIRAJ Review workflow not present on this site — transitions "
          "apply only on sites that carry the workflow fixture")
```

**Expected results:** all four probes print `PASS`. Any assertion failure
is a blocking permission gap — do not tag until it is fixed and the
matrix + this file are re-run.
